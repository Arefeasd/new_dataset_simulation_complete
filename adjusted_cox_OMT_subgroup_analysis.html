Conversion from hospitalization level to patient level

# Convert the dataset from the hospitalization level to the patient level:
# keep only inclusion stays, then retain the earliest inclusion stay
# for each patient according to inclusion_start.
# If several stays have the same earliest date, keep only one of them.
df_patient <- df_large |>
  filter(inclusion_flag == TRUE) |>
  group_by(patient_id) |>
  slice_min(
    order_by = inclusion_start,
    n = 1,
    with_ties = FALSE
  ) |>
  ungroup()

Cardiovascular history variables

# Create patient-level cardiovascular disease indicators from CIM-10 variables,
# then combine them into a composite cardiovascular history variable.
#
# For each disease indicator:
#   1 = the corresponding cardiovascular disease is present,
#   0 = the corresponding cardiovascular disease is absent,
#   NA = the disease status cannot be determined from the available data.
#
# The composite cv_history variable is coded as:
#   1 if at least one cardiovascular condition is present,
#   0 if all cardiovascular conditions are absent,
#   NA if no condition is present but at least one component is missing.

df_patient <- df_patient |>
  mutate(
    ischemic_heart_disease = case_when(
        CIM_10_I20 == 1 |
        CIM_10_I21 == 1 |
        CIM_10_I22 == 1 |
        CIM_10_I25 == 1 ~ 1,
        TRUE ~ 0,
    )
  )


df_patient <- df_patient |>
  mutate(
    atrial_fibrillation = case_when(
        CIM_10_I48 == 1 ~ 1,
        TRUE ~ 0,
    )
  )



df_patient <- df_patient |>
  mutate(
    chronic_heart_failure = case_when(
        CIM_10_I50 == 1 ~ 1,
        TRUE ~ 0,
    )
  )


df_patient <- df_patient |>
  mutate(
    cv_history = case_when(
      ischemic_heart_disease == 1 |
      atrial_fibrillation == 1 |
      chronic_heart_failure == 1 |
      cerebral_vascular_accident_cum == 1 ~ 1,
      TRUE ~ 0
      
    )
  )

Composite severity indicator

# Create a composite severity indicator based on acute respiratory failure
# and the minimum recorded pH value:
#   1 = acute respiratory failure is present or pH is below 7.38,
#   0 = acute respiratory failure is absent or pH is at least 7.38,
#   NA = severity cannot be determined from the available data.

df_patient <- df_patient |>
  mutate(severity = case_when(
    Ins_Resp_Aigue_J96 == 1 | ph_min < 7.38 ~ 1,
    Ins_Resp_Aigue_J96 == 0 | ph_min >= 7.38 ~ 0,
    TRUE ~ NA_real_
  )
)

Patient-level medication indicators

# Create patient-level medication indicators from the corresponding ATC groups
# for beta-blockers, ACE inhibitors, ARBs,
# and aldosterone antagonists/potassium-sparing agents.
#
# For each medication class:
#   1 = at least one related ATC group is present,
#   0 = all related ATC groups are absent,
#   NA = medication use cannot be determined from the available data.

# C07A BETA BLOCKING AGENTS
# C07B BETA BLOCKING AGENTS AND THIAZIDES
# C07C BETA BLOCKING AGENTS AND OTHER DIURETICS
# C07D BETA BLOCKING AGENTS, THIAZIDES AND OTHER DIURETICS
# C07E BETA BLOCKING AGENTS AND VASODILATORS
# C07F BETA BLOCKING AGENTS, OTHER COMBINATIONS

df_patient <- df_patient |>
  mutate(
    beta_blocker = case_when(
        cum_cor_ATC_C07A == 1 |
        cum_cor_ATC_C07B == 1 |
        cum_cor_ATC_C07D == 1 |
        cum_cor_ATC_C07F == 1 ~ 1,
        
        TRUE ~ 0
    )
  )

# C09A ACE INHIBITORS, PLAIN
# C09B ACE INHIBITORS, COMBINATIONS

df_patient <- df_patient |>
  mutate( ACE = 
    case_when(
      cum_cor_ATC_C09A == 1 |
      cum_cor_ATC_C09B == 1 ~ 1,
      
      TRUE ~ 0
    )
  )

# C09C ANGIOTENSIN II RECEPTOR BLOCKERS (ARBs), PLAIN
# C09D ANGIOTENSIN II RECEPTOR BLOCKERS (ARBs), COMBINATIONS

df_patient <- df_patient |>
  mutate( ARB = 
            case_when(
              cum_cor_ATC_C09C == 1 |
              cum_cor_ATC_C09D == 1 ~ 1,
              
              TRUE ~ 0
        )
  )


# C03D ALDOSTERONE ANTAGONISTS AND OTHER POTASSIUM-SPARING AGENTS
df_patient <- df_patient |>
  mutate( Aldosterone = 
            case_when(
                cum_cor_ATC_C03D == 1 ~ 1,
              TRUE ~ 0
            )
  )

Optimal Medical Therapy score

# Create the Optimal Medical Therapy (OMT) score by summing:
#   1 point for beta-blocker use,
#   1 point for either ACE inhibitor or ARB use,
#   1 point for aldosterone antagonist use.
#
# ACE inhibitors and ARBs are treated as one therapeutic component,
# so using both does not add more than one point.
# The resulting OMT score ranges from 0 to 3.
df_patient <- df_patient |>
  mutate(OMT = beta_blocker + pmax(ACE, ARB) + Aldosterone,
         OMT = factor(OMT))
